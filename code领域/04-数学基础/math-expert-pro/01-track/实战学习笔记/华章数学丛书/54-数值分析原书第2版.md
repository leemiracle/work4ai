import numpy as np
from scipy.integrate import odeint
# 1.3 线性一阶：dy/dx + 2y = 0
def f(y, x): return -2 * y
x = np.linspace(0, 5, 100)
y = odeint(f, 1, x)
print(f"y(5) = {y[-1][0]:.6f}, 理论值 e^(-10) = {np.exp(-10):.6f}")
