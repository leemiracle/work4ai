## 💻 实战代码

```python
# V09 模拟简谐振子（Hamilton 力学）
import numpy as np
import matplotlib.pyplot as plt
from scipy.integrate import odeint

# Hamilton 方程：H = p²/2 + q²/2
# dq/dt = p, dp/dt = -q
def hamilton(y, t):
    q, p = y
    return [p, -q]

t = np.linspace(0, 10, 1000)
sol = odeint(hamilton, [1, 0], t)

plt.figure(figsize=(10, 4))
plt.subplot(1, 2, 1)
plt.plot(t, sol[:, 0], label='位置 q')
plt.plot(t, sol[:, 1], label='动量 p')
plt.legend(); plt.title('V09 物理直觉: 简谐振子')

plt.subplot(1, 2, 2)
plt.plot(sol[:, 0], sol[:, 1])
plt.xlabel('q'); plt.ylabel('p')
plt.title('相空间轨迹（圆 = 能量守恒）')
plt.axis('equal')
plt.savefig('V09_physics_intuition.png', dpi=100)
```
