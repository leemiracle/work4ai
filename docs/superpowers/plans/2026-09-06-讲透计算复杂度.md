- [ ] **Step 1: 写解析脚本（标准库 only）**

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""00_zoo_pipeline.py — Complexity Zoo wiki → 结构化数据（图鉴生产的地基）

复杂度动物园(complexityzoo.net)每个类在字母页里是一段
    ===== <span id="xx" ...>类名</span>: 英文全称 =====
    定义与关系原文...
    ----
本脚本把这些段落切成结构化条目并给出分卷方案——演示"非结构化百科→可验证数据"
的通用管线（07 章的活教具）。

用法:
    python 00_zoo_pipeline.py <raw目录> <输出json>
例:
    python 00_zoo_pipeline.py ../../../.research/complexity-zoo/raw \
        ../../../.research/complexity-zoo/zoo_parsed.json
"""
import json, math, re, sys
from pathlib import Path

def strip_tags(s: str) -> str:
    s = re.sub(r'<math>.*?</math>', '', s)              # 行内公式占位
    s = re.sub(r'\[\[[^]|]*\|([^\]]*)\]\]', r'\1', s)   # [[a|b]] -> b
    s = re.sub(r'\[\[([^\]]*)\]\]', r'\1', s)           # [[a]]   -> a
    s = re.sub(r'<sub>', '_', s); s = re.sub(r'<sup>', '^', s)
    s = re.sub(r'<[^>]+>', '', s)
    return re.sub(r'\s+', ' ', s).strip()

HEAD = re.compile(r'^===== <span id="([^"]+)"[^>]*>(.*?)</span>\s*:?\s*(.*?)\s*=+\s*$', re.S)

def parse_page(text: str, letter: str) -> list[dict]:
    out = []
    chunks = re.split(r'^===== ', text, flags=re.M)[1:]
    for ch in chunks:
        m = HEAD.match(ch)
        if not m:
            continue  # 非"类"词条的 ===== 标题（如页首说明），跳过
        body = ch[m.end():]
        body = re.split(r'^----+$', body, maxsplit=1, flags=re.M)[0].strip()
        eid = m.group(1)
        out.append({"id": eid, "letter": letter,
                    "name": strip_tags(m.group(2)), "title": strip_tags(m.group(3)),
                    "body": re.sub(r'\s+', ' ', body)[:4000],
                    "url": f"https://complexityzoo.net/Complexity_Zoo:{letter}#{eid}"})
    return out

def volumes(entries: list[dict]) -> list[tuple[str, int, int, str]]:
    """字母序连续切分，每卷 ~90 类，切点取整字母边界；卷数 6±1。"""
    letters = ["SYM"] + [chr(c) for c in range(ord('A'), ord('Z') + 1)]
    by_l = {l: sum(1 for e in entries if e["letter"] == l) for l in letters}
    by_l = {l: n for l, n in by_l.items() if n}
    total = sum(by_l.values())
    nv, best = 6, None
    for k in (5, 6, 7):
        target, cuts, acc, cur = total / k, [], [], letters[0]
        for l in letters:
            if l not in by_l: continue
            cur_end = l
            acc.append(l)
            if sum(by_l[x] for x in acc) >= target * (len(cuts) + 1) / k * 0.95 and k - len(cuts) - 1 > 0:
                cuts.append(acc[:]); acc = []
        cuts.append(acc[:])
        if len(cuts) == k:
            sizes = [sum(by_l[x] for x in c) for c in cuts]
            score = max(sizes) - min(sizes)
            if best is None or score < best[0]:
                best = (score, cuts)
    _, cuts = best
    vols = []
    for i, c in enumerate(cuts, 1):
        lo = "符号" if c[0] == "SYM" else c[0]
        hi = c[-1] if c[-1] != "SYM" else "符号"
        name = f"卷{i}-{lo}至{hi}.md" if len(c) > 1 else f"卷{i}-{lo}.md"
        if i == 1 and c[0] == "SYM": name = f"卷1-符号与{c[0:1+1]}至{c[-1]}.md"
        vols.append((name, sum(by_l[x] for x in c), len(c), ""))
    return vols

def main():
    raw_dir, out_json = Path(sys.argv[1]), Path(sys.argv[2])
    entries, seen, dupes = [], {}, []
    for f in sorted(raw_dir.glob("Zoo_*.txt")):
        letter = "SYM" if f.stem == "Zoo_SYM" else f.stem.split("_")[1]
        for e in parse_page(f.read_text(encoding="utf-8", errors="replace"), letter):
            key = (e["letter"], e["id"])
            if key in seen: dupes.append(key); continue
            seen[key] = True; entries.append(e)
    order = {l: i for i, l in enumerate(["SYM"] + [chr(c) for c in range(65, 91)])}
    entries.sort(key=lambda e: (order[e["letter"]], e["id"]))
    out_json.write_text(json.dumps(entries, ensure_ascii=False, indent=1), encoding="utf-8")
    print(f"去重总数: {len(entries)}  (重复丢弃: {dupes})")
    per = {}
    for e in entries: per[e["letter"]] = per.get(e["letter"], 0) + 1
    print("逐字母:", {k: per[k] for k in sorted(per, key=lambda x: order.get(x, 99))})
    print("分卷方案(文件名/条数/字母数):")
    for v in volumes(entries): print("  ", v[0], v[1], v[2])
    assert len(entries) > 500, "总数异常，检查抓取完整性"

if __name__ == "__main__":
    main()
```
