# 💻 第 3 幕 · 代码：最小记忆层（200 行 numpy）

把第 2 幕的数学（cosine + 图谱 + Ebbinghaus + RRF）变成可跑代码。

## 核心设计

```python
import numpy as np
from collections import defaultdict
import time

class Memory:
    """最小记忆层: 向量检索 + 知识图谱 + 时间衰减 + RRF 融合."""
    def __init__(self, dim=64, capacity=1000, half_life=3600):
        self.dim = dim
        self.capacity = capacity
        self.half_life = half_life  # Ebbinghaus 半衰期(秒)
        # 向量记忆
        self.vectors = []   # [(id, vec, timestamp, strength)]
        # 知识图谱: 三元组
        self.graph = defaultdict(list)  # head -> [(relation, tail, ts)]
        self.facts = {}      # id -> text
        self._next_id = 0

    def _embed(self, text):
        """简化 embedding: 用 hash 投影成固定向量(仅 demo)."""
        rng = np.random.default_rng(abs(hash(text)) % (2**32))
        return rng.standard_normal(self.dim)

    def add(self, text, triples=None):
        """写入一条记忆 + 可选知识图谱三元组."""
        mid = f"m{self._next_id}"; self._next_id += 1
        vec = self._embed(text)
        self.vectors.append([mid, vec, time.time(), 1.0])
        self.facts[mid] = text
        if triples:
            for h, r, t in triples:
                self.graph[h].append((r, t, time.time()))
        # 容量上限 -> 淘汰最弱的
        if len(self.vectors) > self.capacity:
            self._evict()
        return mid

    def _strength(self, entry):
        """Ebbinghaus 衰减: R(t) = 0.5^(t / half_life), 访问强化 strength."""
        mid, vec, ts, strength = entry
        age = time.time() - ts
        decay = 0.5 ** (age / self.half_life)
        return strength * decay

    def _evict(self):
        """淘汰强度最低的."""
        self.vectors.sort(key=lambda e: self._strength(e))
        removed = self.vectors.pop(0)
        self.facts.pop(removed[0], None)

    def search_vectors(self, query, k=5):
        """向量检索: cosine 相似度 top-k."""
        q = self._embed(query)
        scored = []
        for entry in self.vectors:
            mid, vec, ts, _ = entry
            sim = q @ vec / (np.linalg.norm(q) * np.linalg.norm(vec) + 1e-9)
            scored.append((mid, sim * self._strength(entry)))
        scored.sort(key=lambda x: -x[1])
        return [(mid, rank) for rank, (mid, _) in enumerate(scored[:k])]

    def search_graph(self, head, k=5):
        """图谱检索: 沿关系遍历."""
        edges = self.graph.get(head, [])
        edges.sort(key=lambda e: -e[2])  # 按时间
        return [(f"{head}->{t}", rank) for rank, (_, t, _) in enumerate(edges[:k])]

    def retrieve(self, query, k=5, head=None):
        """RRF 融合: 向量 + 图谱."""
        rankings = [self.search_vectors(query, k)]
        if head:
            rankings.append(self.search_graph(head, k))
        # Reciprocal Rank Fusion
        rrf = defaultdict(float)
        for ranking in rankings:
            for mid, rank in ranking:
                rrf[mid] += 1.0 / (60 + rank)  # k_const=60
        top = sorted(rrf.items(), key=lambda x: -x[1])[:k]
        return [(mid, self.facts.get(mid, mid), score) for mid, score in top]

    def reinforce(self, mid):
        """访问时强化(抗遗忘)."""
        for e in self.vectors:
            if e[0] == mid:
                e[3] += 1.0  # strength+1
                e[2] = time.time()  # 刷新时间
                break


# ===== 跑通验证 =====
if __name__ == "__main__":
    mem = Memory(dim=32, capacity=10)
    # 写入
    mem.add("用户喜欢暗色模式", [("用户", "prefers", "暗色模式")])
    mem.add("用户在写一个 RAG 项目", [("用户", "working_on", "RAG")])
    mem.add("用户用 Python 和 Rust", [("用户", "uses", "Python")])
    time.sleep(0.01)
    mem.add("今天讨论了世界模型")
    # 检索
    print("=== 向量+图谱融合检索 ===")
    for mid, text, score in mem.retrieve("用户的偏好", head="用户"):
        print(f"  [{score:.4f}] {text}")
    print("=== 图谱查询: 用户 uses ? ===")
    for label, rank in mem.search_graph("用户"):
        print(f"  rank {rank}: {label}")
    # 强化 + 再检索
    mem.reinforce("m0")
    print("=== 强化 m0 后 ===")
    for mid, text, score in mem.retrieve("用户喜欢什么"):
        print(f"  [{score:.4f}] {text}")
```

## 这段代码教什么

1. **写入**：`add` 同时存向量 + 图谱三元组
2. **检索**：`retrieve` 用 RRF 融合两路（向量模糊 + 图谱精确）
3. **遗忘**：`_evict` 按衰减强度淘汰——Ebbinghaus 在代码里
4. **强化**：`reinforce` 模拟「访问一次记忆变强」

**生产化要补的**：用真 embedding（sentence-transformers）、HNSW 索引（替代暴力 cosine）、并发安全、持久化。
