## 一、NumPy 裸实现 MRL loss

### 1.1 简化场景：对比损失

给定 batch 的 query 和正样本（batch 内每个 query 配一个 positive），算 MRL 加权对比损失：

```python
import numpy as np

def contrastive_loss(emb_q, emb_p, temperature=0.05):
    """简化 InfoNCE: batch 内其他样本作为负样本.
    emb_q, emb_p: (B, d) 已归一化."""
    sim = emb_q @ emb_p.T / temperature          # (B, B)
    labels = np.arange(emb_q.shape[0])           # 对角线是正样本
    # log-softmax 沿 axis=1
    sim -= sim.max(axis=1, keepdims=True)
    exp = np.exp(sim)
    log_softmax = sim - np.log(exp.sum(axis=1, keepdims=True))
    return -log_softmax[labels, labels].mean()

def mrl_loss(emb_q_full, emb_p_full, matryoshka_dims, weights=None):
    """MRL loss: 在多个嵌套维度上加权求和.
    emb_q_full, emb_p_full: (B, d_full) 已归一化的全维嵌入."""
    d_full = emb_q_full.shape[1]
    assert d_full in matryoshka_dims, f"必须包含全维 {d_full}"
    
    if weights is None:
        weights = [1.0 / len(matryoshka_dims)] * len(matryoshka_dims)
    
    total = 0.0
    for m, w in zip(matryoshka_dims, weights):
        # 截断
        q_m = emb_q_full[:, :m]
        p_m = emb_p_full[:, :m]
        # ★ renorm (必须!)
        q_m = q_m / (np.linalg.norm(q_m, axis=1, keepdims=True) + 1e-12)
        p_m = p_m / (np.linalg.norm(p_m, axis=1, keepdims=True) + 1e-12)
        # base loss
        total += w * contrastive_loss(q_m, p_m)
    return total

# === 跑一下验证 ===
np.random.seed(0)
B, D = 32, 768
matryoshka_dims = [128, 256, 512, 768]

# 模拟 encoder 输出 (batch 内 query 和 positive 应该相似)
emb_q = np.random.randn(B, D); emb_q /= np.linalg.norm(emb_q, axis=1, keepdims=True)
emb_p = emb_q + 0.1 * np.random.randn(B, D)        # positive ≈ query + 小扰动
emb_p /= np.linalg.norm(emb_p, axis=1, keepdims=True)

loss = mrl_loss(emb_q, emb_p, matryoshka_dims)
print(f"MRL loss = {loss:.4f}")  # 应该是个有限的负数
```

### 1.2 反向传播（教学版，NumPy 手写梯度）

完整 MRL 训练要 backward。NumPy 手写 grad 繁琐但教学价值高，这里只示意：

```python
def mrl_loss_and_grad(emb_q, emb_p, dims):
    """教学版: 假设 emb_q, emb_p 是叶子, 求 grad.
    实际训练用 PyTorch autograd."""
    loss = 0
    grad_q = np.zeros_like(emb_q)
    grad_p = np.zeros_like(emb_p)
    for m in dims:
        # forward
        q_m = emb_q[:, :m]; q_m /= np.linalg.norm(q_m, axis=1, keepdims=True) + 1e-12
        p_m = emb_p[:, :m]; p_m /= np.linalg.norm(p_m, axis=1, keepdims=True) + 1e-12
        # ... (省略 contrastive 的 backward)
    return loss, grad_q, grad_p
```

> 实际项目里不要手写 grad，用 PyTorch。

## 三、Matryoshka-Adaptor 实现（给非 MRL 模型升级）

### 3.1 场景

你的 bge-small-zh-v1.5 已经 MNN 部署，想加 MRL 能力，但**不想重训主干**。

### 3.2 完整代码

```python
"""
Matryoshka-Adaptor for bge-small-zh-v1.5
论文: Yoon et al. EMNLP 2024 (arXiv:2407.20243)
依赖: torch, sentence-transformers, numpy
"""
import torch, torch.nn as nn, numpy as np
from sentence_transformers import SentenceTransformer

DEVICE = "cuda" if torch.cuda.is_available() else "cpu"
BASE_MODEL = "/path/to/bge-small-zh-v1.5"      # 你的本地路径
D = 512                                         # bge-small 输出维度
MATRYOSHKA_DIMS = [64, 128, 256, 512]           # 目标截断点

# ========== Adaptor MLP ==========
class MatryoshkaAdaptor(nn.Module):
    def __init__(self, d, hidden=128):
        super().__init__()
        self.fc1 = nn.Linear(d, hidden)
        self.act = nn.ReLU()
        self.fc2 = nn.Linear(hidden, d)
        # ★ 关键: fc2 零初始化 → 训练初始 f(e)=0, ê=e
        nn.init.zeros_(self.fc2.weight)
        nn.init.zeros_(self.fc2.bias)
    def forward(self, e):
        return e + self.fc2(self.act(self.fc1(e)))   # 残差结构

# ========== 冻结主干 ==========
base = SentenceTransformer(BASE_MODEL).to(DEVICE)
for p in base.parameters():
    p.requires_grad = False
base.eval()

adaptor = MatryoshkaAdaptor(D).to(DEVICE)
n_adaptor = sum(p.numel() for p in adaptor.parameters())
n_base = sum(p.numel() for p in base.parameters())
print(f"Adaptor 参数: {n_adaptor:,} (主干 {n_base:,}, 比例 {n_adaptor/n_base*100:.2f}%)")

# ========== Loss 函数 ==========
def l_topk(e_full, ehat_full, dims, topk=10):
    with torch.no_grad():
        sim_full = e_full @ e_full.T
        sim_full.fill_diagonal_(-1)
        _, nn_idx = sim_full.topk(topk, dim=1)
    loss = 0.
    for m in dims:
        ehat_m = ehat_full[:, :m]
        ehat_m = ehat_m / (ehat_m.norm(dim=1, keepdim=True) + 1e-12)
        sim_m = ehat_m @ ehat_m.T
        loss += (sim_m.gather(1, nn_idx) -
                 sim_full.gather(1, nn_idx).detach()).abs().mean()
    return loss / len(dims)

def l_rec(residual):
    return residual.abs().mean()

# ========== 预计算主干 embedding ==========
@torch.no_grad()
def precompute_base_embeddings(sentences, batch=64):
    embs = []
    for i in range(0, len(sentences), batch):
        e = base.encode(sentences[i:i+batch], normalize_embeddings=True,
                        convert_to_tensor=True, device=DEVICE)
        embs.append(e)
    return torch.cat(embs, dim=0)

# ========== 训练循环 ==========
def train_unsupervised(sentences, epochs=5, batch=128, lr=1e-3):
    print(f"预计算 {len(sentences)} 条主干 embedding...")
    E = precompute_base_embeddings(sentences)
    opt = torch.optim.AdamW(adaptor.parameters(), lr=lr)
    N = E.shape[0]
    for ep in range(epochs):
        perm = torch.randperm(N)
        total = 0.
        n_batches = 0
        for i in range(0, N, batch):
            idx = perm[i:i+batch]
            e_batch = E[idx].to(DEVICE)
            residual = adaptor.fc2(adaptor.act(adaptor.fc1(e_batch)))
            ehat = e_batch + residual
            loss = l_topk(e_batch, ehat, MATRYOSHKA_DIMS) + 0.01 * l_rec(residual)
            opt.zero_grad(); loss.backward(); opt.step()
            total += loss.item(); n_batches += 1
        print(f"Epoch {ep+1}: loss = {total/n_batches:.4f}")

# ========== 导出 adaptor 到 ONNX (再转 MNN) ==========
def export_adaptor_onnx(path="adaptor.onnx"):
    adaptor.eval()
    dummy = torch.randn(1, D)
    torch.onnx.export(adaptor, dummy, path,
                      input_names=["e"], output_names=["ehat"],
                      dynamic_axes={"e": {0: "batch"}, "ehat": {0: "batch"}})
    print(f"Adaptor 导出到 {path}")

# ========== 运行 ==========
# sentences = [...]  # 你的中文语料 (10k+ 条)
# train_unsupervised(sentences)
# export_adaptor_onnx("adaptor.onnx")
# 然后用 onnx2mnn 转 MNN, 接在 bge-small-zh-MNN 后面
```

### 3.3 部署链（端侧）

```
[文本] → [bge-small-zh-v1.5-MNN]   → e (512d)
                                       ↓
                                 [adaptor.mnn]   → ê (512d, 已有 MRL 性质)
                                       ↓
                                 [截断到 128 + renorm]
                                       ↓
                                 [sqlite-vec 入库]
```

Adaptor 参数量约 130K（512×128 + 128×512），MNN Q4 量化后约 200 KB。**端侧开销几乎为零**。

## 四、Permutation 方法（替代 Adaptor）

Yoon 等在 [primeqa/docuverse](https://github.com/primeqa/docuverse) 还实现了 Permutation 方法：

**思想**：不学习变换，只学习一个**维度重排**——把重要的维度排到前面。

```python
class ScoreVectorPermutation(nn.Module):
    """学一个分数向量 s ∈ R^d, argsort(s) 就是重排索引"""
    def __init__(self, d):
        super().__init__()
        self.score = nn.Parameter(torch.zeros(d))
    def forward(self, e):
        # 训练时: NeuralSort 可微近似
        # 推理时: 直接 argsort + gather
        idx = torch.argsort(-self.score)
        return e[:, idx]
```

**优势**：
- 参数量 $d$（如 768 个标量）
- **完全保留原 cosine 相似度**（重排不改变向量内积）
- 与现有索引兼容（不需要重建库，只需重排）

**劣势**：
- 表达能力弱于 MLP Adaptor
- 不能"修改"原嵌入，只能重排

### 5.2 超参建议

| 超参 | 推荐值 | 备注 |
|---|---|---|
| Adaptor hidden $h$ | $d/4$（如 512→128） | 太大易过拟合 |
| Learning rate | 1e-3（无监督）/ 1e-4（监督） | AdamW |
| Batch size | 128-256 | 小 batch 会让 top-k 邻居样本少 |
| Epochs | 5（无监督）+ 10（监督） | 早停 |
| $\alpha$（L_pair） | 0.1 | 论文默认 |
| $\beta$（L_rec） | 0.01 | 防止 adaptor 改得太狠 |
| $\gamma$（L_rank） | 1.0 | 监督阶段 |

### 5.3 训练时间（CPU 参考）

| 阶段 | 数据量 | 单 epoch | 总时间 |
|---|---|---|---|
| 无监督 | 10k 句 | ~10s | ~1 分钟 |
| 无监督 | 100k 句 | ~100s | ~10 分钟 |
| 监督 | 5k 三元组 | ~5s | ~1 分钟 |
